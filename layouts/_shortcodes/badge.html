{{/* A shortcode to create a badge. @param {string} content The content of the badge. @param {string} color The color of the badge. @param {string} class The class of the badge. @param {string} link The link of the badge. @param {string} icon The icon of the badge. A bundled or remote icon name, or `file:` for an SVG in the page bundle or the assets directory. @param {string} size The size of the badge. One of xs, sm, md, lg, xl. Default md. @deprecated type Earlier name for `color`; still accepted. @param {bool} border Draw the badge outline. Default true; pass false for a flat badge. or @param {string} 0 The content of the badge. @example {{< badge content="Badge" color="blue" >}} @example {{< badge content="Badge" icon="file:icons/hexagon.svg" size="lg" >}} @example {{< badge content="Badge" icon="iconify:devicon-plain/homebrew" >}} @example {{< badge "Badge" >}} */}} {{- if .IsNamedParams -}} {{- $content := .Get "content" -}} {{- $color := .Get "color" | default (.Get "type") | default "" -}}{{- /* Compatibility with previous parameter. */ -}} {{- $class := .Get "class" | default "" -}} {{- $link := .Get "link" | default "" -}} {{- $icon := .Get "icon" | default "" -}} {{- $size := .Get "size" | default "md" -}} {{- /* Not `not (eq X false) | default true`: `default` treats false as empty, so that form always returned true and the documented parameter did nothing. Comparing the formatted value is nil-safe - a missing parameter prints ``, which is not "false", so the default stays true. */ -}} {{- $border := ne (printf "%v" (.Get "border")) "false" -}} {{- if $link -}} {{- /* rel="noreferrer" because target="_blank" without it leaks the referrer and window.opener. */ -}} {{- partial "shortcodes/badge.html" (dict "content" $content "color" $color "class" $class "border" $border "icon" $icon "size" $size "page" $.Page ) -}} {{- else -}} {{- partial "shortcodes/badge.html" (dict "content" $content "color" $color "class" $class "border" $border "icon" $icon "size" $size "page" $.Page ) -}} {{- end -}} {{- else -}} {{- $content := .Get 0 -}} {{- partial "shortcodes/badge.html" (dict "content" $content "border" true "page" $.Page ) -}} {{- end -}}
