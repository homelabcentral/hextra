{{/* Resolve the blog layout configuration for a page into a single dict. Every feature is opt-in: defining the corresponding block under `params.blog` in the site configuration turns it on, and an explicit `enable: false` turns it back off. A site with no `params.blog` gets a configuration where everything is disabled, which renders the pre-existing blog markup unchanged. Config keys are lowercased by Hugo, so presence checks use lowercase names. @param {page} . The page being rendered. @returns {dict} rail, widgets, list, card, article @example {{ $cfg := partial "blog/config.html" . }} */}} {{- $page := . -}} {{/* Every block below is walked with `index`, which fails the build on a scalar - and since `navbar.html` reaches this partial through `blog/has-rail.html`, that failure would take every page with it, not only the blog. `blog/as-map.html` warns once per path and hands back an empty dict, so a malformed block disables its own feature and nothing else. Each level needs its own call: guarding `params.blog` alone still left `blog.list: 10` to fail on the first `index` inside it. */}} {{- $blog := partial "blog/as-map.html" (dict "value" site.Params.blog "name" "blog") -}} {{/* ---------------------------------------------------------------- rail */}} {{- $rail := dict "enable" false -}} {{- with partial "blog/as-map.html" (dict "value" $blog.rail "name" "blog.rail") -}} {{- $rail = dict "enable" (not (eq (index . "enable") false)) "onArticle" (eq (index . "onarticle") true) "profile" (index . "profile") "sponsor" (index . "sponsor") -}} {{- end -}} {{/* Per-page opt-out: `blog: { rail: false }` in front matter */}} {{- with $page.Params.blog -}} {{- if reflect.IsMap . -}} {{- if eq (index . "rail") false -}} {{- $rail = merge $rail (dict "enable" false) -}} {{- end -}} {{- end -}} {{- end -}} {{/* ------------------------------------------------------------- widgets */}} {{- $widgets := dict "enable" false -}} {{- with partial "blog/as-map.html" (dict "value" $blog.widgets "name" "blog.widgets") -}} {{- $config := . -}} {{- $sections := dict -}} {{- range $name, $default := (dict "recent" 5 "pinned" 3 "tags" 10) -}} {{- $section := dict "enable" false "count" $default -}} {{/* `recent: 5` rather than `recent: { count: 5 }` is the obvious way to write this and the wrong one, so it goes through the guard too. */}} {{- with partial "blog/as-map.html" (dict "value" (index $config $name) "name" (printf "blog.widgets.%s" $name)) -}} {{- $section = dict "enable" (not (eq (index . "enable") false)) "count" (int ((index . "count") | default $default)) -}} {{- end -}} {{- $sections = merge $sections (dict $name $section) -}} {{- end -}} {{- $widgets = merge (dict "enable" (not (eq (index $config "enable") false))) $sections -}} {{- end -}} {{/* ---------------------------------------------------------------- list */}} {{/* The blog list's own settings, as opposed to the card it renders. These exist here rather than being read straight off `site.Params.blog.list` at the call sites because that is what made a malformed `params.blog` fail the build: `blog/list.html`, `term.html` and `blog/post-card.html` each walked into it themselves, so sanitising it in this partial protected nothing. Values are passed through rather than coerced. `sortBy` and `sortOrder` are `utils/sort-pages`' to validate, and `displayTags` is read as a plain truthiness test by its one caller - narrowing either here would change behaviour rather than centralise it. */}} {{- $listConfig := partial "blog/as-map.html" (dict "value" $blog.list "name" "blog.list") -}} {{- $list := dict "sortBy" nil "sortOrder" nil "pagerSize" 10 "displayTags" nil -}} {{- with $listConfig -}} {{- $list = dict "sortBy" (index . "sortby") "sortOrder" (index . "sortorder") "pagerSize" (int ((index . "pagersize") | default 10)) "displayTags" (index . "displaytags") -}} {{- end -}} {{/* ---------------------------------------------------------------- card */}} {{/* `layout` is the blog list, `termLayout` the tag pages. The two cards are one partial and are freely interchangeable, so the defaults here are the only reason each page looks the way it always has. Reached through `$listConfig` rather than `$blog.list.card`, so a scalar `blog.list` is a warning above rather than a field lookup on an int here. */}} {{- $layouts := slice "vertical" "horizontal" -}} {{- $card := dict "enable" false "cover" false "readingTime" false "layout" "vertical" "termLayout" "horizontal" -}} {{- with partial "blog/as-map.html" (dict "value" $listConfig.card "name" "blog.list.card") -}} {{- $enable := not (eq (index . "enable") false) -}} {{- $layout := (index . "layout") | default "vertical" -}} {{- if not (in $layouts $layout) -}} {{- warnidf "blog-card-layout" `blog: params.blog.list.card.layout is %q; expected "vertical" or "horizontal". Falling back to "vertical".` $layout -}} {{- $layout = "vertical" -}} {{- end -}} {{- $termLayout := (index . "termlayout") | default "horizontal" -}} {{- if not (in $layouts $termLayout) -}} {{- warnidf "blog-card-termlayout" `blog: params.blog.list.card.termLayout is %q; expected "vertical" or "horizontal". Falling back to "horizontal".` $termLayout -}} {{- $termLayout = "horizontal" -}} {{- end -}} {{- $card = dict "enable" $enable "cover" (and $enable (not (eq (index . "cover") false))) "readingTime" (and $enable (not (eq (index . "readingtime") false))) "layout" $layout "termLayout" $termLayout -}} {{- end -}} {{/* ------------------------------------------------------------- article */}} {{- $article := dict "cover" false "readingTime" false "tags" false "displayPagination" true "share" (dict "enable" false "links" slice) "related" (dict "enable" false "count" 3) "series" (dict "enable" false "opened" false) -}} {{- with partial "blog/as-map.html" (dict "value" $blog.article "name" "blog.article") -}} {{- $articleConfig := . -}} {{- $share := dict "enable" false "links" slice -}} {{- with partial "blog/as-map.html" (dict "value" (index $articleConfig "share") "name" "blog.article.share") -}} {{- $share = dict "enable" (and (not (eq (index . "enable") false)) (gt (len ((index . "links") | default slice)) 0)) "links" ((index . "links") | default slice) -}} {{- end -}} {{- $related := dict "enable" false "count" 3 -}} {{- with partial "blog/as-map.html" (dict "value" (index $articleConfig "related") "name" "blog.article.related") -}} {{- $related = dict "enable" (not (eq (index . "enable") false)) "count" (int ((index . "count") | default 3)) -}} {{- end -}} {{/* Presence is the switch, so the check is `isset` rather than `with`: `series: {}` and a bare `series:` are empty values that `with` would skip, leaving the block an author wrote with no effect. Only an explicit false - either `series: false` or `enable: false` - opts back out. A scalar is meaningful here, so this one is not routed through `blog/as-map.html`. */}} {{- $series := dict "enable" false "opened" false -}} {{- if isset $articleConfig "series" -}} {{- $value := index $articleConfig "series" -}} {{- if reflect.IsMap $value -}} {{- $series = dict "enable" (not (eq (index $value "enable") false)) "opened" (eq (index $value "opened") true) -}} {{- else -}} {{- $series = dict "enable" (ne $value false) "opened" false -}} {{- end -}} {{- end -}} {{/* `displayPagination` is compared against `false` rather than run through `default`, which is what the call site in `blog/single.html` used to do. Hugo treats `false` as empty, so `default true false` is `true` and the documented `displayPagination: false` silently did nothing. */}} {{- $article = dict "cover" (eq (index $articleConfig "cover") true) "readingTime" (eq (index $articleConfig "readingtime") true) "tags" (eq (index $articleConfig "tags") true) "displayPagination" (not (eq (index $articleConfig "displaypagination") false)) "share" $share "related" $related "series" $series -}} {{- end -}} {{- return (dict "rail" $rail "widgets" $widgets "list" $list "card" $card "article" $article) -}}
