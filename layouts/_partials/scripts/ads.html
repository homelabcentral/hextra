{{/* Ad network loaders. Emitted only on pages that actually rendered an ad, gated on the "hasAds" store flag in scripts.html, and only for the networks that page used - a post with one EthicalAds unit never fetches AdSense. Carbon is absent on purpose. It writes its ad at the position of its own script element, so `ads/slot.html` emits that inline at the slot instead. When `verifyAllPages` is on, `ads/verify.html` has already put a loader in the head of every page - but only for the site-level `adsense.client`. Skipping this block outright therefore left every other publisher id on the page with no loader at all, so a slot or an ad shortcode naming a second client rendered an ad unit that nothing could fill. Only the id the head already covers is dropped here; two loaders for one publisher id is the configuration Google warns about, one loader per id is not. */}} {{- $providers := .Store.Get "hextraAdProviders" | default slice -}} {{- $ads := partialCached "ads/config.html" . .RelPermalink -}} {{- $inHead := false -}} {{- $headClient := "" -}} {{- with index $ads.providers "adsense" -}} {{- $inHead = index . "verifyallpages" -}} {{- if $inHead -}}{{- $headClient = index . "client" | default "" -}}{{- end -}} {{- end -}} {{- $clients := .Store.Get "hextraAdClients" | default slice -}} {{- if $inHead -}} {{- $pending := slice -}} {{- range $clients -}} {{- if ne . $headClient -}}{{- $pending = $pending | append . -}}{{- end -}} {{- end -}} {{- $clients = $pending -}} {{- end -}} {{- if in $providers "adsense" -}} {{- if not $inHead -}} {{- end -}} {{- /* One loader per publisher id in use; see ads/slot.html. */ -}} {{- range $client := $clients -}} {{- end -}} {{- end -}} {{- if in $providers "ethicalads" -}} {{- end -}}
